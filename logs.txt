# Buat Key Root CA
openssl genrsa -out ca.key 4096

# Buat Sertifikat Root CA
openssl req -x509 -new -nodes -key ca.key -sha256 -days 3650 -out ca.crt


# Buat Private Key WEB1
openssl genrsa -out web1.key 2048

# Buat CSR (Permintaan Sertifikat)
openssl req -new -key web1.key -out web1.csr

# Tandatangani dengan Root CA tadi
openssl x509 -req -in web1.csr -CA ca.crt -CAkey ca.key -CAcreateserial \
-out web1.crt -days 365 -sha256

# Pindahkan file ke tempat yang aman
sudo mv web1.crt /etc/ssl/certs/
sudo mv web1.key /etc/ssl/private/
sudo mv ca.crt /etc/ssl/certs/

# Atur Izin Akses (Penting!)
sudo chmod 600 /etc/ssl/private/web1.key
